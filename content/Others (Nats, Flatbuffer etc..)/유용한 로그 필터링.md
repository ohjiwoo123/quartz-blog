5. dirn5454 필터링해서 segment_idx가 같은 partial_idx가 0인 timestamp와 segment_timestamp가 다르면 MISMATCH로 표시!!
```
grep 'dirn5454' server_2026-08-20.aslog |
grep -E '"stream_type":[[:space:]]*1([^0-9]|$)' |
awk '
/send media segment/ {
    match($0, /"segment_idx":[[:space:]]*([0-9]+)/, a)
    match($0, /"segment_timestamp":[[:space:]]*([0-9]+)/, b)

    if (a[1] != "" && b[1] != "")
        media[a[1]] = b[1]

    next
}

/send partial segment/ {
    match($0, /"segment_idx":[[:space:]]*([0-9]+)/, a)
    match($0, /"partial_idx":[[:space:]]*([0-9]+)/, p)
    match($0, /"partial_timestamp":[[:space:]]*([0-9]+)/, b)

    if (a[1] != "" && p[1] == "0" && b[1] != "")
        partial[a[1]] = b[1]
}

END {
    total = 0
    match_count = 0
    mismatch = 0

    for (idx in media) {
        if (idx in partial) {
            total++

            if (media[idx] == partial[idx]) {
                match_count++
            } else {
                mismatch++

                print "MISMATCH idx=" idx \
                      " media=" media[idx] \
                      " partial=" partial[idx]
            }
        }
    }

    print "=============================="
    print "TOTAL    =" total
    print "MATCH    =" match_count
    print "MISMATCH =" mismatch
    print "=============================="
}
'
```
